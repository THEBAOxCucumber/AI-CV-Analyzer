ตัวอย่างการใช้สำหรับ Admin เท่านั้น:


role.middleware.ts
router.get(
  "/admin-only",
  authenticateToken,
  authorizeRoles("ADMIN"),
  adminController,
);

หรืออนุญาตทั้งสอง Role:

router.get(
  "/users-and-admins",
  authenticateToken,
  authorizeRoles("USER", "ADMIN"),
  controller,
);
